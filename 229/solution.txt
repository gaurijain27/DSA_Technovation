class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        int c1=0;
        int c2=0;
        int count1=0;
        int count2=0;
        
        int limit = nums.size()/3;
        vector<int>ans;

        for(int c:nums){
            if(c==c1)count1++;
            else if(c==c2)count2++;
            else if(count1==0){
                c1=c;
                count1=1;
            }
            else if(count2==0){
                c2=c;
                count2=1;
            }
            else{
                count1--;
                count2--;
            }
        }
        count1=count2=0;

        for (int c : nums) {
            if (c == c1) {
                count1++;
            } else if (c == c2) {
                count2++;
            }
        }
         if (count1 > limit) {
            ans.push_back(c1);
        }
        if (count2 > limit) {
            ans.push_back(c2);
        }

        return ans;   
    }
};
